2. `create_alignment` name=ELM-AVE, begin_station="10+00", start=[1000, 3000], legs=[{bearing "S 80°00'00\" E", 800}, {deflection "40 L", radius 500, 700}, {deflection "25 R", radius 700, 600}].

4. `draw_offset` alignment=ELM-AVE, offsets=[-11, 11], layer=rd_EP.
5. `draw_offset` alignment=ELM-AVE, offsets=[-16, 16], layer=rd_ES.
6. `annotate_stationing` space=model, alignment=ELM-AVE, interval=100, layer=rd_LAYOUT-NOTE-anno.
